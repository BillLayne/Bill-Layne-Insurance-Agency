# Rule of thumb (Google Search Central): robots.txt controls CRAWLING, not
# indexing. A page that must stay out of Google gets a noindex meta tag and
# must remain crawlable so Googlebot can see that tag. Never add a Disallow
# for a page that carries noindex - blocking + noindex lets the URL be indexed
# from links with no snippet.

User-agent: *
Allow: /

# Non-content data directories only. CSS and JS stay crawlable so Google can
# render mobile layouts and interactive sections exactly as users see them.
Disallow: /data/
Disallow: /auto-center/Logos/
